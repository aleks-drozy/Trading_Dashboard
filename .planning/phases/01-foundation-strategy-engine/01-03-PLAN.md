**data/bar_store.py**:
```python
from dataclasses import dataclass
from datetime import datetime
from typing import Any
import threading

@dataclass
class Bar:
    timestamp: datetime
    open: float
    high: float
    low: float
    close: float
    volume: float

class BarStore:
    """Thread-safe in-memory store: symbol -> list of recent closed bars."""
    def __init__(self):
        self._data: dict[str, list[Bar]] = {}
        self._lock = threading.Lock()

    def update(self, symbol: str, bars: list[Bar]) -> None:
        with self._lock:
            self._data[symbol] = bars

    def get(self, symbol: str) -> list[Bar]:
        with self._lock:
            return self._data.get(symbol, [])

    def symbols(self) -> list[str]:
        with self._lock:
            return list(self._data.keys())

# Module-level singleton — imported by feeds and strategy engine
bar_store = BarStore()
```

**data/binance_feed.py** — implement Pattern 4 from research (Binance WebSocket with Proactive 23-Hour Restart):

```python
import asyncio, logging
from datetime import datetime, timezone, timedelta
from binance import AsyncClient, BinanceSocketManager
from backend.data.bar_store import bar_store, Bar

RECONNECT_INTERVAL_SECONDS = 23 * 3600
WATCHDOG_TIMEOUT_SECONDS = 180  # 3 minutes

logger = logging.getLogger(__name__)

class BinanceFeed:
    def __init__(self, symbols: list[str] | None = None):
        self.symbols = symbols or ["BTCUSDT"]
        self._last_bar_time: dict[str, datetime] = {}

    def _on_closed_bar(self, kline: dict) -> None:
        """Process a closed kline message into BarStore."""
        symbol = kline["s"]
        bar = Bar(
            timestamp=datetime.fromtimestamp(kline["t"] / 1000, tz=timezone.utc),
            open=float(kline["o"]),
            high=float(kline["h"]),
            low=float(kline["l"]),
            close=float(kline["c"]),
            volume=float(kline["v"]),
        )
        current = bar_store.get(symbol)
        bar_store.update(symbol, (current + [bar])[-500:])  # keep last 500 bars
        self._last_bar_time[symbol] = bar.timestamp
        logger.info("Binance closed bar: %s @ %s close=%s", symbol, bar.timestamp, bar.close)

    def _check_watchdog(self, symbol: str) -> None:
        last = self._last_bar_time.get(symbol)
        if last and (datetime.now(timezone.utc) - last) > timedelta(seconds=WATCHDOG_TIMEOUT_SECONDS):
            logger.error("WATCHDOG: No bar received for %s in >3 minutes. Last: %s", symbol, last)

    async def run(self) -> None:
        """Outer loop: proactive 23-hour restart."""
        while True:
            try:
                client = await AsyncClient.create(api_key=None, api_secret=None)
                bm = BinanceSocketManager(client)
                # Subscribe to all symbols (one socket per symbol for simplicity)
                for symbol in self.symbols:
                    async with bm.kline_socket(symbol, interval="1m") as stream:
                        deadline = asyncio.get_event_loop().time() + RECONNECT_INTERVAL_SECONDS
                        while asyncio.get_event_loop().time() < deadline:
                            msg = await asyncio.wait_for(stream.recv(), timeout=WATCHDOG_TIMEOUT_SECONDS)
                            if msg is None or msg.get("e") == "error":
                                logger.warning("Binance stream error for %s, restarting", symbol)
                                break
                            kline = msg["k"]
                            self._check_watchdog(symbol)
                            if kline["x"]:  # closed bar only
                                self._on_closed_bar(kline)
            except asyncio.TimeoutError:
                logger.error("Binance stream timeout — no message in %ds", WATCHDOG_TIMEOUT_SECONDS)
            except Exception as e:
                logger.error("Binance feed error: %s — restarting in 5s", e)
                await asyncio.sleep(5)
            finally:
                try:
                    await client.close_connection()
                except Exception:
                    pass
            await asyncio.sleep(1)

# Module-level singleton
binance_feed = BinanceFeed()
```

```python
from backend.data.binance_feed import binance_feed
from backend.data.yfinance_feed import poll_yfinance_loop
from backend.watchlist.repository import WatchlistRepository

@asynccontextmanager
async def lifespan(app: FastAPI):
    create_db_and_tables()
    with Session(engine) as session:
        seed_defaults(session)
    # Start data feeds
    binance_task = asyncio.create_task(binance_feed.run())
    def get_watchlist_symbols() -> list[str]:
        with Session(engine) as s:
            return [w.symbol for w in WatchlistRepository(s).get_all()]
    yfinance_task = asyncio.create_task(poll_yfinance_loop(get_watchlist_symbols))
    yield
    binance_task.cancel()
    yfinance_task.cancel()
    await asyncio.gather(binance_task, yfinance_task, return_exceptions=True)
```
