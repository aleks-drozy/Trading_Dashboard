    df = pd.read_csv("tests/fixtures/spy_1min_tv_reference.csv", parse_dates=["timestamp"])
    df = df.set_index("timestamp")
    # Apply engine to all rows

    mismatches = (ref != computed).sum()

**Lookahead bias test (in test_engine.py):**
```python
def test_no_lookahead_bias():
    df = pd.read_csv("tests/fixtures/spy_1min_tv_reference.csv", parse_dates=["timestamp"])
    df = df.set_index("timestamp").iloc[:100]  # 100-bar slice
    # Run engine on original
    result_a = StrategyEngine().run(df)
    # Tamper the last bar (the open/current bar that should be excluded)
    df_tampered = df.copy()
    df_tampered.iloc[-1, df_tampered.columns.get_loc("close")] = df_tampered.iloc[-1]["close"] * 10  # extreme change
    result_b = StrategyEngine().run(df_tampered)
    # Result must be identical — last bar must have been excluded
    assert result_a == result_b, "Strategy result changed when open bar was tampered — lookahead bias present"
```
